        function setBilling(cycle) {
            currentCycle = cycle;
            const slider = document.getElementById('billing-slider');
            const btns = ['monthly', 'halfYear', 'yearly'];
            
            btns.forEach((btn, index) => {
                const el = document.getElementById('btn-' + btn);
                if (btn === cycle) {
                    el.classList.replace('text-slate-500', 'text-slate-900');
                    if(index === 0) slider.style.transform = 'translateX(0)';
                    else if(index === 1) slider.style.transform = 'translateX(100%)';
                    else slider.style.transform = 'translateX(200%)';
                } else {
                    el.classList.replace('text-slate-900', 'text-slate-500');
                }
            });

            updatePrice('basic', pricingData[cycle].basic, cycle);
            updatePrice('pro', pricingData[cycle].pro, cycle);
            updatePrice('advanced', pricingData[cycle].adv, cycle);
        }

        function updatePrice(plan, price, cycle) {
            const priceEl = document.getElementById(`price-${plan}`);
            const periodEl = document.getElementById(`period-${plan}`);
            const strikeEl = document.getElementById(`strike-${plan}`);
            const badgeEl = document.getElementById(`save-badge-${plan}`);
            
            if(!priceEl) return;

            priceEl.style.opacity = '0';
            if(strikeEl) strikeEl.style.opacity = '0';
            if(badgeEl) badgeEl.style.opacity = '0';

            setTimeout(() => {
                priceEl.innerText = '฿' + price.toLocaleString();
                
                if (cycle === 'monthly') {
                    periodEl.innerText = '/เดือน';
                    if(strikeEl) strikeEl.innerText = '';
                } else {
                    periodEl.innerText = cycle === 'halfYear' ? '/6 เดือน' : '/ปี';
                    
                    const months = cycle === 'halfYear' ? 6 : 12;
                    const fullPrice = pricingData.monthly[plan] * months;
                    const savings = fullPrice - price;
                    
                    if(strikeEl) {
                        strikeEl.innerText = '฿' + fullPrice.toLocaleString();
                        strikeEl.style.opacity = '1';
                    }
                    if(badgeEl) {
                        badgeEl.innerText = 'ประหยัด ฿' + savings.toLocaleString();
                        badgeEl.style.opacity = '1';
                    }
                }
                
                priceEl.style.opacity = '1';
            }, 150);
        }

        function openPartnerAssessment() {
            currentAnswers = {};
            assessmentModal.classList.remove('opacity-0', 'pointer-events-none');
            setTimeout(() => { assessmentContent.classList.remove('scale-95', 'opacity-0'); }, 10);
            renderQuiz();
        }

        function closePartnerAssessment() {
            assessmentContent.classList.add('scale-95', 'opacity-0');
            setTimeout(() => { assessmentModal.classList.add('opacity-0', 'pointer-events-none'); }, 300);
            
            // Reset views
            setTimeout(() => {
                document.getElementById('quiz-container').classList.remove('hidden');
                document.getElementById('quiz-footer').classList.remove('hidden');
                document.getElementById('quiz-result').classList.add('hidden');
            }, 300);
        }

        function renderQuiz() {
            const container = document.getElementById('quiz-questions');
            let html = '';
            
            questionsData.forEach((qData, qIndex) => {
                html += `<div class="bg-slate-50 p-6 rounded-2xl border border-slate-100">
                    <p class="font-bold text-slate-900 mb-4 leading-relaxed">${qData.q}</p>
                    <div class="space-y-2">`;
                
                qData.options.forEach((opt, oIndex) => {
                    const id = `q${qIndex}_o${oIndex}`;
                    html += `
                        <label class="flex items-start gap-3 p-3 rounded-xl border border-slate-200 bg-white cursor-pointer hover:border-indigo-300 transition-colors">
                            <input type="radio" name="q${qIndex}" value="${opt.s}" class="mt-1 w-4 h-4 text-indigo-600 focus:ring-indigo-500" onclick="answerQuestion(${qIndex}, ${opt.s})" />
                            <span class="text-sm text-slate-700">${opt.t}</span>
                        </label>
                    `;
                });
                html += `</div></div>`;
            });
            
            container.innerHTML = html;
        }

        function answerQuestion(qIndex, score) {
            currentAnswers[qIndex] = score;
        }

        function submitAssessment() {
            const totalQuestions = questionsData.length;
            const answeredCount = Object.keys(currentAnswers).length;

            if (answeredCount < totalQuestions) {
                alert(`กรุณาตอบคำถามให้ครบทุกข้อ (ขาดอีก ${totalQuestions - answeredCount} ข้อ)`);
                return;
            }

            const btn = document.getElementById('btn-submit-quiz');
            const originalHTML = btn.innerHTML;
            btn.innerHTML = '<div class="w-5 h-5 border-2 border-white/30 border-t-white rounded-full animate-spin"></div> กำลังประมวลผล...';
            btn.classList.add('pointer-events-none', 'opacity-80');

            setTimeout(() => {
                btn.innerHTML = originalHTML;
                btn.classList.remove('pointer-events-none', 'opacity-80');
                showResult();
            }, 1500);
        }

        function showResult() {
            let totalScore = 0;
            for (let key in currentAnswers) {
                totalScore += currentAnswers[key];
            }
            
            // 50% passing score (Max is 100, pass is 50)
            const passed = totalScore >= 50;

            document.getElementById('quiz-container').classList.add('hidden');
            document.getElementById('quiz-footer').classList.add('hidden');
            
            const resultBox = document.getElementById('quiz-result');
            resultBox.classList.remove('hidden');

            const iconBox = document.getElementById('result-icon');
            const title = document.getElementById('result-title');
            const scoreDisplay = document.getElementById('result-score');
            const desc = document.getElementById('result-desc');
            const action = document.getElementById('result-action');

            scoreDisplay.innerHTML = `คะแนนของคุณ: <span class="${passed ? 'text-emerald-500' : 'text-rose-500'}">${totalScore}/100</span>`;

            if (passed) {
                iconBox.className = 'w-24 h-24 rounded-full flex items-center justify-center mb-6 bg-emerald-100 text-emerald-500';
                iconBox.innerHTML = '<i data-lucide="check-circle-2" class="w-12 h-12"></i>';
                title.innerText = 'ยินดีด้วย คุณผ่านเกณฑ์!';
                title.className = 'text-2xl font-black text-emerald-600 mb-2';
                desc.innerText = 'คุณมีวิสัยทัศน์และศักยภาพที่พร้อมสำหรับการเป็นตัวแทนจำหน่าย AIVA ขั้นตอนต่อไปคือการสมัครสร้างบัญชี Partner อย่างเป็นทางการครับ';
                action.innerHTML = `
                    <button onclick="window.location.reload()" class="w-full bg-slate-900 hover:bg-emerald-600 text-white font-bold py-3.5 rounded-xl transition-all shadow-sm flex items-center justify-center gap-2">
                        ไปที่หน้าสมัคร Partner <i data-lucide="arrow-right" class="w-4 h-4"></i>
                    </button>
                `;
            } else {
